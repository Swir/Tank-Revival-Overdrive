name: ROADMAP v13.6 Qualification Finalizer

on:
  workflow_run:
    workflows: ['Battlefield Sensor Fusion v13.6 Windows Qualification Gate']
    types: [completed]

permissions:
  contents: write
  actions: read

jobs:
  finalize:
    if: ${{ github.event.workflow_run.conclusion == 'success' && github.event.workflow_run.head_branch == 'dev-v13-6' }}
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - name: Capture exact qualified candidate
        shell: bash
        run: |
          set -euo pipefail
          echo "CANDIDATE_SHA=${{ github.event.workflow_run.head_sha }}" >> "$GITHUB_ENV"
          echo "QUALIFICATION_RUN=${{ github.event.workflow_run.id }}" >> "$GITHUB_ENV"
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.event.workflow_run.head_sha }}
          fetch-depth: 0
      - name: Refuse stale or non-current candidate
        shell: bash
        run: |
          set -euo pipefail
          test "$(git rev-parse HEAD)" = "${CANDIDATE_SHA}"
          git fetch origin dev-v13-6
          current="$(git rev-parse origin/dev-v13-6)"
          test "$current" = "${CANDIDATE_SHA}" || { echo "Qualified SHA is stale: current=$current candidate=$CANDIDATE_SHA"; exit 1; }
          test "$(cat VERSION)" = 'v13.6.0-dev'
      - name: Revalidate open qualified source state
        shell: bash
        run: |
          set -euo pipefail
          python .github/scripts/test_package_v136_sensor_fusion.py
          python Tools/verify_v136_sensor_fusion.py
          python Tools/verify_csharp_type_uniqueness.py
          python Tools/verify_production_audio_assets.py
          python Tools/verify_unity_ci_bootstrap.py
          python Tools/generate_progress_svgs.py --check
          python Tools/swir_visual_report_guard.py --check
      - name: Finalize only the verified v13.6 checklist and dashboard
        shell: python
        run: |
          from pathlib import Path
          import os,re
          p=Path('ROADMAP.md')
          s=p.read_text(encoding='utf-8')
          heading='## v13.6 — Battlefield Sensor Fusion & Contact Warfare — IN DEVELOPMENT'
          if heading not in s:
              raise SystemExit('open v13.6 heading missing')
          prefix, section = s.split(heading,1)
          if section.count('- [ ]') != 8 or section.count('- [x]') != 0:
              raise SystemExit('v13.6 finalizer requires exactly 8 open items')
          section=section.replace('- [ ]','- [x]')
          s=prefix+'## v13.6 — Battlefield Sensor Fusion & Contact Warfare — QUALIFIED'+section
          s=s.replace('ROADMAP-98.2%25-blue','ROADMAP-100.0%25-brightgreen',1)
          s=s.replace('DONE-431%2F439-1f6feb','DONE-439%2F439-1f6feb',1)
          s=s.replace('STATUS-V13.6%20IN%20DEVELOPMENT-blue','STATUS-V13.6%20QUALIFIED-brightgreen',1)
          s=s.replace('Roadmap progress: 431 / 439 completed (98.2%) — V13.6 IN DEVELOPMENT.',
                      'Roadmap progress: 439 / 439 completed (100.0%) — V13.6 QUALIFIED.',1)
          s=s.replace('| **431** | **8** | **439** | **98.2%** |',
                      '| **439** | **0** | **439** | **100.0%** |',1)
          done=len(re.findall(r'^- \[x\] ',s,re.M)); open_=len(re.findall(r'^- \[ \] ',s,re.M))
          if (done,open_,done+open_) != (439,0,439):
              raise SystemExit(f'closed roadmap counts invalid: {(done,open_,done+open_)}')
          p.write_text(s,encoding='utf-8')

          q=Path('docs/qualification/V13_6_WINDOWS_QUALIFICATION.md')
          q.parent.mkdir(parents=True,exist_ok=True)
          q.write_text(
              '# v13.6 Windows Qualification Evidence\n\n'
              f'- Candidate SHA: `{os.environ["CANDIDATE_SHA"]}`\n'
              f'- Qualification workflow run: `{os.environ["QUALIFICATION_RUN"]}`\n'
              '- Gate: `Battlefield Sensor Fusion v13.6 Windows Qualification Gate`\n'
              '- Result: `SUCCESS`\n'
              '- Matrix: production authored-audio preflight; v13.6 sensor fusion; v13.5 weather; v13.4 tactical terrain; v13.3 cohesion; rounds 80/90/100 soak.\n'
              '- Release note: milestone qualification is not the same as a public GitHub Release.\n',
              encoding='utf-8')
      - name: Regenerate and verify SVG-only qualified presentation
        shell: bash
        run: |
          set -euo pipefail
          python Tools/generate_progress_svgs.py --write --embed --check
          python Tools/swir_visual_report_guard.py --check
          python Tools/generate_progress_svgs.py --write --embed --check
          python Tools/swir_visual_report_guard.py --check
          python - <<'PY'
          from pathlib import Path
          import re
          s=Path('ROADMAP.md').read_text(encoding='utf-8')
          done=len(re.findall(r'^- \[x\] ',s,re.M)); open_=len(re.findall(r'^- \[ \] ',s,re.M))
          assert (done,open_,done+open_)==(439,0,439)
          assert '| **439** | **0** | **439** | **100.0%** |' in s
          assert 'V13.6%20QUALIFIED' in s
          assert s.count('assets/readme/progress-mini.svg')==1
          r=Path('README.md').read_text(encoding='utf-8')
          assert r.count('assets/readme/progress-card.svg')==1
          print('v13.6 closed-state roadmap/SVG contract: PASS')
          PY
          git diff --check
      - name: Commit exact-SHA qualification finalization
        shell: bash
        run: |
          set -euo pipefail
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add ROADMAP.md README.md assets/readme/progress-card.svg assets/readme/progress-mini.svg assets/readme/progress-template.svg docs/qualification/V13_6_WINDOWS_QUALIFICATION.md
          git commit -m "docs(v13.6): finalize exact-SHA Windows qualification"
          git push origin "HEAD:dev-v13-6"
